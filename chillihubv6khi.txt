-- Hàm kiểm tra Key đã lưu ẩn trong máy
local function checkSavedKey()
    if isfile and readfile and isfile(KEY_FILE_NAME) then
        local success, content = pcall(readfile, KEY_FILE_NAME)
        if success and content then
            local savedKey, savedTime = string.match(content, "^(.-)|(%d+)$")
            if savedKey and savedTime then
                savedTime = tonumber(savedTime)
                if savedKey == CORRECT_KEY and (os.time() - savedTime) < EXPIRATION_SECONDS then
                    return true
                end
            end
        end
    end
    return false
end

-- Hàm lưu Key ngầm vào thiết bị
local function saveKeyLocally(key)
    if writefile then
        pcall(function()
            writefile(KEY_FILE_NAME, key .. "|" .. tostring(os.time()))
        end)
    end
end

local function showNotification(text, textColor, strokeColor, bgColor)
    StatusLabel.Text = text
    StatusLabel.TextColor3 = textColor
    BadgeStroke.Color = strokeColor
    StatusBadge.BackgroundColor3 = bgColor
    
    TweenService:Create(StatusBadge, TweenInfo.new(0.25), {BackgroundTransparency = 0.2}):Play()
    TweenService:Create(BadgeStroke, TweenInfo.new(0.25), {Transparency = 0.3}):Play()

end

-- Hiệu ứng Lắc Mượt
local isShaking = false
local function smoothShake()
    if isShaking then return end
    isShaking = true
    
    local originalPos = UDim2.new(0.5, 0, 0.5, 0)
    local offsets = {10, -8, 6, -4, 2, 0}
    local tweenInfo = TweenInfo.new(0.05, Enum.EasingStyle.Quad, Enum.EasingDirection.Out)
    
    for _, offset in ipairs(offsets) do
        local targetPos = originalPos + UDim2.new(0, offset, 0, 0)
        local tween = TweenService:Create(Container, tweenInfo, {Position = targetPos})
        tween:Play()
        tween.Completed:Wait()
    end
    
    isShaking = false
end
